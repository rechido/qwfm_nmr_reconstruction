# NMR 복원 실험: 측정점 64개, 스펙트럼 128구간
# 이 파일이 있는 프로그램 최상위 폴더에서 필요한 명령을 하나씩 실행합니다.
# 원하는 모델의 학습·평가 명령을 선택합니다. 전체 모델 비교 시에는 모든 모델을 실행합니다.
# A: SMC 참조분포 비교는 1 → 2 → 3A → 4A → 5 순서입니다.
# B: 합성 정답 기반 ES/KS 평가는 1 → 2 → 3B → 4B 순서이며, 그림도 자동 생성됩니다.
# 두 평가를 모두 수행할 수 있습니다. 기존 수치해법만 평가할 때는 2번 학습을 생략합니다.

# 1. 실행 환경 준비
# Linux/Bash에서 CUDA 호환 PyTorch가 설치된 Python 환경을 활성화합니다.
# 학습 및 학습 모델의 표본 생성은 GPU 4개 설정입니다. 실행 스크립트의 GPU_IDS를 확인합니다.
python -m pip install -r requirements.txt

# 2. 모델 학습
# 학습 데이터는 실행 중 자동 생성됩니다. 별도 데이터 다운로드나 평가 데이터 준비는 필요 없습니다.
# 모델별 학습이 완료된 뒤 해당 모델을 평가합니다. 동일 설정의 학습을 마쳤으면 이 단계를 생략합니다.
# HFM
bash scripts/m64n128/01_train_histogram_linear.sh
# QWFM (L=128)
bash scripts/m64n128/02_train_quantile_wasserstein_l128.sh
# QWFM (L=256)
bash scripts/m64n128/03_train_quantile_wasserstein_l256.sh
# QWFM (L=512)
bash scripts/m64n128/04_train_quantile_wasserstein_l512.sh
# 회귀 (MSE)
bash scripts/m64n128/05_train_deterministic_regression.sh
# 조건부 정규화 흐름
bash scripts/m64n128/07_train_conditional_normalizing_flow.sh
# 조건부 VAE (z=16)
bash scripts/m64n128/08_train_conditional_vae_z16.sh
# 회귀 (MSE+W1)
bash scripts/m64n128/09_train_deterministic_regression_mse_w1.sh
# 조건부 VAE (z=32)
bash scripts/m64n128/10_train_conditional_vae_z32.sh

# 3A. SMC 비교용 평가 데이터와 참조분포 준비
# 세 참조분포 생성을 모두 마친 뒤 00d를 실행합니다. 검사에 실패하면 4A로 진행하지 않습니다.
bash scripts/m64n128/00a_build_posterior_reference_run0.sh
bash scripts/m64n128/00b_build_posterior_reference_run1.sh
bash scripts/m64n128/00c_build_posterior_reference_run2.sh
bash scripts/m64n128/00d_audit_posterior_references.sh

# 3B. ES/KS 평가용 합성 관측·정답 데이터 준비
# 선택한 실험에서 한 번 실행합니다. SMC 참조분포나 학습된 모델은 필요 없습니다.
bash scripts/m64n128/proper_scores/00_prepare_test_conditions.sh

# 4A. SMC 참조분포를 이용한 복원·평가

# 학습 모델의 01은 해당 체크포인트가 필요합니다. 기존 수치해법의 01과 모든 02는 CPU에서 실행합니다.
